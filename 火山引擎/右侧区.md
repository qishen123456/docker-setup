# 导入必要的库
import pandas as pd

# 合并df5和df7的数据，计算环比增长率和贡献度
def calculate_growth_and_contribution(current_df: pd.DataFrame, previous_df: pd.DataFrame) -> pd.DataFrame:
    """
    计算各分公司的环比增长率和贡献度
    
    参数:
        current_df: 当前交易日数据(2025-05-20)
        previous_df: 上一交易日数据(2025-05-13)
        
    返回:
        包含计算结果的数据框
    """
    # 重命名列名以保持一致性
    previous_df = previous_df.rename(columns={
        '交易总金额(求和)': '上期交易总金额',
        '交易ID(去重计数)': '上期交易总笔数'
    })
    current_df = current_df.rename(columns={
        '交易总金额(求和)': '本期交易总金额',
        '交易总笔数': '本期交易总笔数'
    })
    
    # 合并两期数据
    merged_df = pd.merge(
        current_df, 
        previous_df, 
        on='所属分公司', 
        how='outer'
    ).fillna(0)  # 处理可能存在的分公司缺失情况
    
    # 计算总交易金额和总交易笔数的变化
    total_current_amount = merged_df['本期交易总金额'].sum()
    total_previous_amount = merged_df['上期交易总金额'].sum()
    total_current_count = merged_df['本期交易总笔数'].sum()
    total_previous_count = merged_df['上期交易总笔数'].sum()
    
    # 计算环比增长率
    merged_df['交易总金额环比增长率'] = (
        (merged_df['本期交易总金额'] - merged_df['上期交易总金额']) / 
        merged_df['上期交易总金额'].replace(0, float('nan'))
    ).fillna(0)  # 避免除以0的情况
    
    merged_df['交易总笔数环比增长率'] = (
        (merged_df['本期交易总笔数'] - merged_df['上期交易总笔数']) / 
        merged_df['上期交易总笔数'].replace(0, float('nan'))
    ).fillna(0)
    
    # 计算贡献度
    merged_df['交易总金额贡献度'] = (
        (merged_df['本期交易总金额'] - merged_df['上期交易总金额']) / 
        (total_current_amount - total_previous_amount)
    ).fillna(0)
    
    merged_df['交易总笔数贡献度'] = (
        (merged_df['本期交易总笔数'] - merged_df['上期交易总笔数']) / 
        (total_current_count - total_previous_count)
    ).fillna(0)
    
    # 选择需要的列并返回结果
    result_df = merged_df[[
        '所属分公司',
        '交易总金额环比增长率',
        '交易总笔数环比增长率',
        '交易总金额贡献度',
        '交易总笔数贡献度'
    ]]
    
    return result_df
